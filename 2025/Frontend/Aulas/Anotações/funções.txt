let pessoas = [
    {
        nome: "caio",
        idade: 20,
        filhos: [
            { nome: "joao" },
            { nome: "bianca" }
        ]
    },
    {
        nome: "joelinton",
        idade: 20,
        filhos: [
            { nome: "lucas" },
            { nome: "jose" }
        ]
    }
]; -> exemplo de json ou objeto
console.log(pessoas[1].filhos[0].nome); -> exibe lucas

let filhos = pessoas.flatMap(pessoa => pessoa.filhos);
console.log(filhos);
// Saída: [{ nome: "joao" }, { nome: "bianca" }, { nome: "lucas" }, { nome: "jose" }]

let lucas = pessoas
    .flatMap(pessoa => pessoa.filhos) // Primeiro, achata o array de filhos
    .find(filho => filho.nome === "lucas"); // Busca o primeiro filho chamado "lucas"

console.log(lucas);
// Saída: { nome: "lucas" }

let primeiroFilho = pessoas.find(pessoa => pessoa.nome === "joelinton").filhos[0].nome;

console.log(primeiroFilho); // Saída: "lucas"

let filhosJoelinton = pessoas.find(pessoa => pessoa.nome === "joelinton").filhos;

filhosJoelinton.forEach(filho => {
    console.log(filho.nome); // Mostra o nome de cada filho
});

// Alternativamente, exibir todos os filhos diretamente:
console.log(filhosJoelinton);
// Saída: [{ nome: "lucas" }, { nome: "jose" }]
